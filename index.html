const { useState, useEffect, useRef, useCallback } = React;


const LOAN_TYPES = ["30yr Fixed","15yr Fixed","ARM 5/1","FHA","VA","Jumbo","HELOC"];

const BRANCHES   = ["Chicago-Loop","NYC-Midtown","LA-Downtown","Austin-North","Miami-Beach","Seattle-SLU","Dallas-Uptown","Boston-Back Bay"];

const OFFICERS   = ["A.Johnson","B.Kim","C.Patel","D.Rivera","E.Smith","F.Chen","G.Torres","H.Nguyen"];

const CHANNELS   = ["Online","Branch","Mobile","Broker","Referral"];

const STAGES     = ["Application Intake","Identity Verification","Credit Pull","Income Verification","Underwriting","Appraisal","Offer Generated"];

const FRAUD_SIG  = ["duplicate_ssn","suspicious_device","mismatched_address","velocity_check"];

const ANOMALIES  = ["credit_bureau_timeout","doc_parse_failure","underwriter_queue_spike","appraisal_vendor_error","identity_api_slow"];

const STAGE_CLR  = ["#6366f1","#8b5cf6","#a78bfa","#06b6d4","#f59e0b","#ef4444","#22c55e"];

const FIRSTS     = ["James","Maria","Robert","Lisa","David","Sarah","Michael","Jennifer","William","Jessica"];

const LASTS      = ["Anderson","Martinez","Johnson","Williams","Brown","Davis","Wilson","Taylor","Moore","Jackson"];

const DC         = {APPROVE:"#22c55e",REVIEW:"#f59e0b",REJECT:"#ef4444"};


const rand    = a  => a[Math.floor(Math.random()*a.length)];

const randInt = (lo,hi) => Math.floor(Math.random()*(hi-lo+1))+lo;

const randF   = (lo,hi) => +(Math.random()*(hi-lo)+lo).toFixed(2);

const fmt$    = n  => "$"+n.toLocaleString();

const fmtMs   = ms => ms>=1000?(ms/1000).toFixed(1)+"s":ms+"ms";

const getTier = s  => s>=720?"Prime":s>=660?"Near-Prime":"Sub-Prime";

const decide  = (score,dti,hasFraud) => {

  if(hasFraud) return "REJECT";

  if(score>720&&dti<0.35) return "APPROVE";

  if(score<620||dti>0.50) return "REJECT";

  return "REVIEW";

};

function hotStage(anomaly,i){

  return (anomaly==="credit_bureau_timeout"&&i===2)||(anomaly==="underwriter_queue_spike"&&i===4)||

         (anomaly==="appraisal_vendor_error"&&i===5)||(anomaly==="identity_api_slow"&&i===1)||

         (anomaly==="doc_parse_failure"&&i===3);

}


let _uid=1;

function buildLoan(injectAnomaly,injectFraud){

  const score=randInt(560,840),dti=randF(0.18,0.58),ltv=randF(0.55,0.98);

  const fraud=injectFraud?rand(FRAUD_SIG):"none";

  const anomaly=injectAnomaly?rand(ANOMALIES):"none";

  const stageLat=STAGES.map((_,i)=>{

    let b=[90,400,900,500,2200,1600,200][i];

    if(anomaly==="credit_bureau_timeout"&&i===2)b=randInt(9000,16000);

    if(anomaly==="underwriter_queue_spike"&&i===4)b=randInt(14000,28000);

    if(anomaly==="appraisal_vendor_error"&&i===5)b=randInt(11000,22000);

    if(anomaly==="identity_api_slow"&&i===1)b=randInt(5000,12000);

    if(anomaly==="doc_parse_failure"&&i===3)b=randInt(4000,8000);

    return Math.max(40,b+randInt(-80,200));

  });

  const totalDur=stageLat.reduce((a,b)=>a+b,0);

  const loanType=rand(LOAN_TYPES),branch=rand(BRANCHES),officer=rand(OFFICERS),channel=rand(CHANNELS);

  const amount=randInt(85000,2800000);

  const decision=decide(score,dti,fraud!=="none");

  const status=(anomaly!=="none"&&Math.random()>0.45)||fraud!=="none"?"error":"success";

  return {

    id:"LN-"+String(_uid++).padStart(5,"0"),

    traceId:Math.random().toString(16).slice(2).padEnd(32,"0"),

    applicant:rand(FIRSTS)+" "+rand(LASTS),

    applicantId:"APP-"+randInt(10000,99999),

    ts:new Date(),loanType,branch,officer,channel,amount,

    score,dti,ltv,fraud,anomaly,stageLat,totalDur,status,decision,

    tier:getTier(score),

    riskScore:Math.round(100-score/10+dti*80+ltv*30+(fraud!=="none"?40:0)),

    _sent:false,

  };

}


function loanToEvents(l){

  const rootId=Math.random().toString(16).slice(2,18);

  let off=0;

  const evts=[{time:l.ts.toISOString(),data:{

    name:"loan.origination","trace.trace_id":l.traceId,"trace.span_id":rootId,

    duration_ms:l.totalDur,status:l.status,decision:l.decision,

    "loan.type":l.loanType,"loan.amount_usd":l.amount,"loan.officer":l.officer,

    "branch.id":l.branch,channel:l.channel,"applicant.id":l.applicantId,

    "applicant.credit_score":l.score,"applicant.credit_tier":l.tier,

    "applicant.dti_ratio":l.dti,"applicant.ltv_ratio":l.ltv,

    "risk.score":l.riskScore,"fraud.signal":l.fraud,"anomaly.type":l.anomaly,

    "service.name":"loan-origination-platform",

  }}];

  STAGES.forEach((stage,i)=>{

    const sid=Math.random().toString(16).slice(2,18),d=l.stageLat[i],isHot=hotStage(l.anomaly,i);

    evts.push({time:new Date(l.ts.getTime()+off).toISOString(),data:{

      name:stage.toLowerCase().replace(/ /g,"_"),"trace.trace_id":l.traceId,

      "trace.span_id":sid,"trace.parent_id":rootId,duration_ms:d,

      "stage.name":stage,"stage.index":i,"stage.is_anomalous":isHot,

      status:isHot&&l.status==="error"?"error":"ok",

      "loan.type":l.loanType,"branch.id":l.branch,"loan.officer":l.officer,

      "service.name":"loan-origination-platform",

    }});

    off+=d;

  });

  return evts;

}
